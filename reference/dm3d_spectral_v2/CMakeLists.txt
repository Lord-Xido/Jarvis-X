cmake_minimum_required(VERSION 3.16)
project(jarvis_x_dm3d_spectral_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
find_path(BOOST_MULTIPRECISION_INCLUDE_DIR boost/multiprecision/cpp_int.hpp REQUIRED)
add_executable(jarvis_x_dm3d_v2 jarvis_x_dm3d_v2.cpp)
target_include_directories(jarvis_x_dm3d_v2 PRIVATE ${BOOST_MULTIPRECISION_INCLUDE_DIR})
target_link_libraries(jarvis_x_dm3d_v2 PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(jarvis_x_dm3d_v2 PRIVATE /W4 /O2)
else()
    target_compile_options(jarvis_x_dm3d_v2 PRIVATE -Wall -Wextra -Wpedantic -O3)
endif()
enable_testing()
add_test(NAME dm3d_v2_self_test COMMAND jarvis_x_dm3d_v2 --self-test)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME dm3d_v2_integration
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:jarvis_x_dm3d_v2>)
endif()
