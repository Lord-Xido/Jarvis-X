cmake_minimum_required(VERSION 3.16)
project(jarvisx_dm3d_8k3_virtual LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(jarvisx_dm3d_8k3_virtual jarvis_x_8k_virtual.cpp)
if(MSVC)
  target_compile_options(jarvisx_dm3d_8k3_virtual PRIVATE /W4)
else()
  target_compile_options(jarvisx_dm3d_8k3_virtual PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME jx8k_cpp_selftest COMMAND jarvisx_dm3d_8k3_virtual --self-test)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME jx8k_independent_python
  COMMAND ${Python3_EXECUTABLE} -m unittest -v test_stack.py)
set_tests_properties(jx8k_independent_python PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
