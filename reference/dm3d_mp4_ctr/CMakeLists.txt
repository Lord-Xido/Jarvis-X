cmake_minimum_required(VERSION 3.16)
project(jarvisx_dm3d_mp4_ctr_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(dm3d-mp4-ctr dm3d_mp4_ctr.cpp)
if(MSVC)
  target_compile_options(dm3d-mp4-ctr PRIVATE /W4)
else()
  target_compile_options(dm3d-mp4-ctr PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
enable_testing()
add_test(NAME dm3d_mp4_ctr_selftest COMMAND dm3d-mp4-ctr selftest)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME dm3d_mp4_ctr_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_integration.py $<TARGET_FILE:dm3d-mp4-ctr>)
endif()
