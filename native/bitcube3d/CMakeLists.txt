cmake_minimum_required(VERSION 3.16)
project(bitcube3d VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(bitcube3d_sdk STATIC src/engine.cpp)
target_include_directories(bitcube3d_sdk PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_library(bitcube3d_program STATIC src/program.cpp)
target_link_libraries(bitcube3d_program PUBLIC bitcube3d_sdk)
add_executable(bitcube3d-cpp src/coding_cli.cpp)
target_link_libraries(bitcube3d-cpp PRIVATE bitcube3d_program)
if(MSVC)
  target_compile_options(bitcube3d_sdk PRIVATE /W4)
  target_compile_options(bitcube3d_program PRIVATE /W4)
else()
  target_compile_options(bitcube3d_sdk PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(bitcube3d_program PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(bitcube3d_integration tests/integration.cpp)
  target_link_libraries(bitcube3d_integration PRIVATE bitcube3d_program)
  add_test(NAME bitcube3d_integration COMMAND bitcube3d_integration)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME bitcube3d_cli_smoke
      COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py" "$<TARGET_FILE:bitcube3d-cpp>")
  endif()
endif()
