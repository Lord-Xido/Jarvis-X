cmake_minimum_required(VERSION 3.16)
project(jx64_autoencoding3d VERSION 2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(jx64_core src/engine.cpp)
target_include_directories(jx64_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jx64_core PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
add_executable(jx64_engine src/main.cpp)
target_link_libraries(jx64_engine PRIVATE jx64_core)
add_executable(jx64_tests tests/test_engine.cpp)
target_link_libraries(jx64_tests PRIVATE jx64_core)
enable_testing()
add_test(NAME jx64_reference_vs_separable COMMAND jx64_tests)
