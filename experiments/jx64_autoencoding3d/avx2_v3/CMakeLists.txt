cmake_minimum_required(VERSION 3.16)
project(jx64_avx2_fma LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CheckCXXCompilerFlag)
add_compile_options(-O3 -mavx2 -mfma -fno-fast-math -Wall -Wextra -Wpedantic)
add_executable(jx64avx2 jx64_avx2_fma.cpp)
add_executable(jx64test test_correctness.cpp)
enable_testing()
add_test(NAME jx64_correctness COMMAND jx64test)
