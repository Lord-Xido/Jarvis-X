cmake_minimum_required(VERSION 3.16)
project(jx3d_codec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(jx3d jx3d_codec.cpp)
target_compile_options(jx3d PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(jx3d PRIVATE ZLIB::ZLIB OpenSSL::Crypto)
