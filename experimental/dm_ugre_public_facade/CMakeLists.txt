cmake_minimum_required(VERSION 3.16)
project(dm_ugre_public_facade LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(dm_ugre_facade src/main.cpp)
if(MSVC)
  target_compile_options(dm_ugre_facade PRIVATE /W4 /WX)
else()
  target_compile_options(dm_ugre_facade PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()
enable_testing()
add_test(NAME facade_invariants COMMAND dm_ugre_facade --self-test)
add_test(NAME facade_demo COMMAND dm_ugre_facade --demo)
set_tests_properties(facade_demo PROPERTIES PASS_REGULAR_EXPRESSION "RESULT fixture/r1 Synthetic report")
