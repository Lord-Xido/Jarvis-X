cmake_minimum_required(VERSION 3.18)
project(jx_gcl_8k LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(gcl8k main.cpp)
target_compile_options(gcl8k PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
include(CheckLanguage)
option(GCL_ENABLE_CUDA "Build optional NVIDIA CUDA fused renderer" ON)
if(GCL_ENABLE_CUDA)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    target_sources(gcl8k PRIVATE gcl_cuda.cu)
    target_compile_definitions(gcl8k PRIVATE GCL_HAS_CUDA=1)
    set_target_properties(gcl8k PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON CUDA_SEPARABLE_COMPILATION ON)
  else()
    message(STATUS "CUDA toolkit not present: building deterministic CPU reference")
  endif()
endif()
enable_testing()
add_test(NAME gcl8k_cpu_self_test COMMAND gcl8k --self-test)
add_test(NAME gcl8k_cpu_rectangular COMMAND gcl8k --width 19 --height 17)
