cmake_minimum_required(VERSION 3.16)
project(JarvisXJoint3D LANGUAGES CXX)
find_path(ZSTD_INCLUDE_DIR zstd.h REQUIRED)
find_library(ZSTD_LIBRARY NAMES zstd REQUIRED)
add_executable(jx3d_joint jx3d_joint.cpp)
target_compile_features(jx3d_joint PRIVATE cxx_std_17)
target_link_libraries(jx3d_joint PRIVATE zstd)
if(MSVC)
  target_compile_options(jx3d_joint PRIVATE /W4 /WX)
else()
  target_compile_options(jx3d_joint PRIVATE -O3 -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
enable_testing()
add_test(NAME selftest COMMAND jx3d_joint --self-test)
set_tests_properties(selftest PROPERTIES TIMEOUT 60)
