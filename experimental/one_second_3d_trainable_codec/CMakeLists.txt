cmake_minimum_required(VERSION 3.16)
project(jx3d_one_second LANGUAGES CXX)
add_executable(jx3d jarvis_x_3d_one_second.cpp)
target_compile_features(jx3d PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(jx3d PRIVATE /W4 /WX)
else()
  target_compile_options(jx3d PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
enable_testing()
add_test(NAME jx3d_selftest COMMAND jx3d --self-test)
add_test(NAME jx3d_one_second COMMAND jx3d --budget-ms 1000 --epochs 120)
set_tests_properties(jx3d_selftest jx3d_one_second PROPERTIES TIMEOUT 10)
