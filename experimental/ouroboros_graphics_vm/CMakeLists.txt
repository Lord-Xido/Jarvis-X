cmake_minimum_required(VERSION 3.16)
project(jarvis_x_ouroboros_graphics_vm LANGUAGES CXX)
add_executable(jx_ouroboros_vm jarvis_x_3d_vm.cpp)
target_compile_features(jx_ouroboros_vm PRIVATE cxx_std_17)
if (MSVC)
  target_compile_options(jx_ouroboros_vm PRIVATE /W4 /WX)
else()
  target_compile_options(jx_ouroboros_vm PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
enable_testing()
add_test(NAME ouroboros_vm_selftest COMMAND jx_ouroboros_vm --self-test)
set_tests_properties(ouroboros_vm_selftest PROPERTIES TIMEOUT 60)
