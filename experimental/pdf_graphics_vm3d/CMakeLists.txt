cmake_minimum_required(VERSION 3.16)
project(jx_pdf_graphics_vm3d LANGUAGES CXX)
add_executable(jx_pdf_graphics_vm3d main.cpp)
target_compile_features(jx_pdf_graphics_vm3d PRIVATE cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jx_pdf_graphics_vm3d PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
enable_testing()
add_test(NAME pdf3d_self_test COMMAND jx_pdf_graphics_vm3d --self-test)
