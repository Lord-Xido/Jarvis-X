cmake_minimum_required(VERSION 3.16)
project(jx_fractal_chain LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(jx_fractal_chain main_fractal_runtime.cpp)
if(MSVC)
 target_compile_options(jx_fractal_chain PRIVATE /W4)
else()
 target_compile_options(jx_fractal_chain PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME fractal_chain_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_contract.py $<TARGET_FILE:jx_fractal_chain>)
add_executable(jx_fractal_quality quality_probe.cpp)
if(NOT MSVC)
 target_compile_options(jx_fractal_quality PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME fractal_fixed_reference_quality
 COMMAND jx_fractal_quality ${CMAKE_CURRENT_BINARY_DIR}/fractal_quality.csv)
