cmake_minimum_required(VERSION 3.16)
project(jx_latent512 LANGUAGES CXX)
add_executable(jxl512 jx_latent512.cpp)
target_compile_features(jxl512 PRIVATE cxx_std_17)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jxl512 PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
enable_testing()
add_test(NAME jxl512_selftest COMMAND jxl512 selftest)
add_test(NAME jxl512_demo COMMAND jxl512 demo)
