cmake_minimum_required(VERSION 3.16)
project(jarvisx_nano3d_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(jarvisx_nano3d nano3d.cpp)
if(MSVC)
  target_compile_options(jarvisx_nano3d PRIVATE /W4 /permissive-)
else()
  target_compile_options(jarvisx_nano3d PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
add_executable(jarvisx_nano3d_sphere sphere_optimizer.cpp)
if(MSVC)
  target_compile_options(jarvisx_nano3d_sphere PRIVATE /W4 /permissive-)
else()
  target_compile_options(jarvisx_nano3d_sphere PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME nano3d_invariants COMMAND $<TARGET_FILE:jarvisx_nano3d> --no-emit --loops 10000)
  add_test(NAME nano3d_one_shot COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_one_shot.py ${CMAKE_CXX_COMPILER})
  set_tests_properties(nano3d_one_shot PROPERTIES TIMEOUT 300)
  add_test(NAME nano3d_transactions COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_reference.py --engine $<TARGET_FILE:jarvisx_nano3d> --compiler ${CMAKE_CXX_COMPILER})
endif()
