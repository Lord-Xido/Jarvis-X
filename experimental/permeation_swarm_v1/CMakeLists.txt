cmake_minimum_required(VERSION 3.16)
project(jx3d_permeation_swarm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_executable(jx3d_permeation_swarm main.cpp)
target_link_libraries(jx3d_permeation_swarm PRIVATE OpenMP::OpenMP_CXX)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jx3d_permeation_swarm PRIVATE -O2 -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME physics_and_determinism COMMAND jx3d_permeation_swarm --self-test)
add_test(NAME smoke_replay COMMAND jx3d_permeation_swarm --swarm 2 --steps 2 --threads 2 --seed 42)
