cmake_minimum_required(VERSION 3.16)
project(moagi_ann3d_inward_loop LANGUAGES CXX)
add_executable(moagi_ann3d moagi_ann3d.cpp)
target_compile_features(moagi_ann3d PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(moagi_ann3d PRIVATE /W4)
else()
    target_compile_options(moagi_ann3d PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME moagi_ann3d_selftest COMMAND moagi_ann3d --self-test)
