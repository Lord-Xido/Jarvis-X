cmake_minimum_required(VERSION 3.16)
project(JX3D_Fractal_Cloud_Reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(jx3d_cloud src/jx3d_cloud.cpp)
target_compile_options(jx3d_cloud PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(jx3d_cloud PRIVATE Threads::Threads)
