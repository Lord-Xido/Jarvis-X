cmake_minimum_required(VERSION 3.16)
project(JX_PixelROM100 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PNG REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_executable(jx_pixelrom jx_pixelrom.cpp)
target_link_libraries(jx_pixelrom PRIVATE PNG::PNG OpenSSL::Crypto)
add_executable(jx_vm jx_mp_1e12_pixelrom.cpp)
target_link_libraries(jx_vm PRIVATE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(jx_pixelrom PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(jx_vm PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME vm_selftest COMMAND jx_vm --self-test)

find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME pixelrom_100banks_roundtrip_and_tamper
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/integration_test.py
            ${CMAKE_CURRENT_BINARY_DIR} $<TARGET_FILE:jx_pixelrom> $<TARGET_FILE:jx_vm>)
endif()
