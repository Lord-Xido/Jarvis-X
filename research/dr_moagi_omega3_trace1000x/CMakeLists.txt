cmake_minimum_required(VERSION 3.16)
project(MoagiOmega3Trace1000x LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)
add_executable(moagi_trace1000x src/moagi_trace1000x.cpp)
target_link_libraries(moagi_trace1000x PRIVATE OpenSSL::Crypto)
if(MSVC)
  target_compile_options(moagi_trace1000x PRIVATE /W4)
else()
  target_compile_options(moagi_trace1000x PRIVATE -O3 -Wall -Wextra)
endif()
enable_testing()
add_test(NAME trace1000x_selftest COMMAND moagi_trace1000x selftest)
