cmake_minimum_required(VERSION 3.16)
project(dm3d_isa_omega LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(dm3d_isa dm3d_isa.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(dm3d_isa PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME dm3d_self_test COMMAND dm3d_isa --self-test)
add_test(NAME dm3d_demo COMMAND dm3d_isa --demo)
add_test(NAME dm3d_quine COMMAND dm3d_isa --quine)
