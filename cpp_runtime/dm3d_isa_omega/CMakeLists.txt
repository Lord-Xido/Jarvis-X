cmake_minimum_required(VERSION 3.18)
project(dm3d_isa_omega LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(dm3d_isa dm3d_isa.cpp)
add_executable(dm3d_pipeline dm3d_pipeline.cpp)
target_link_libraries(dm3d_pipeline PRIVATE Threads::Threads)
foreach(target dm3d_isa dm3d_pipeline)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
# Optional CUDA backend; builds on hosts with a CUDA compiler. No GPU required
# for the CPU regression suite. CUDA parity must be validated on a GPU host.
option(DM3D_ENABLE_CUDA "Try to build optional CUDA block-average backend" ON)
if(DM3D_ENABLE_CUDA)
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    add_library(dm3d_cuda STATIC cuda_block.cu)
    set_target_properties(dm3d_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_definitions(dm3d_pipeline PRIVATE DM3D_HAS_CUDA=1)
    target_link_libraries(dm3d_pipeline PRIVATE dm3d_cuda)
  endif()
endif()
enable_testing()
add_test(NAME dm3d_self_test COMMAND dm3d_isa --self-test)
add_test(NAME dm3d_demo COMMAND dm3d_isa --demo)
add_test(NAME dm3d_quine COMMAND dm3d_isa --quine)
add_test(NAME dm3d_v2_unit COMMAND dm3d_pipeline --self-test-v2)
add_test(NAME dm3d_v2_cuda_capability COMMAND dm3d_pipeline --cuda-check)
add_test(NAME dm3d_v2_integration COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/test_integration.sh" "$<TARGET_FILE:dm3d_pipeline>")
set_tests_properties(dm3d_v2_integration PROPERTIES TIMEOUT 120)
