cmake_minimum_required(VERSION 3.16)
project(DrMoagi3D VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(dm3d_core STATIC src/core.cpp src/selftest.cpp)
target_include_directories(dm3d_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(dm3d_core PUBLIC Threads::Threads)
if(MSVC)
  target_compile_options(dm3d_core PRIVATE /W4)
else()
  target_compile_options(dm3d_core PRIVATE -Wall -Wextra -O2)
endif()
add_executable(dm3d src/main.cpp)
target_link_libraries(dm3d PRIVATE dm3d_core)
enable_testing()
add_executable(dm3d_tests tests/test_engine.cpp)
target_link_libraries(dm3d_tests PRIVATE dm3d_core)
add_test(NAME engine_roundtrip COMMAND dm3d_tests)
