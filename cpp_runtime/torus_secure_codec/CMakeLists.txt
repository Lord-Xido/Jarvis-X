cmake_minimum_required(VERSION 3.16)
project(jarvisx_torus_secure_codec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL 1.1.1 REQUIRED COMPONENTS Crypto)
add_executable(torus_secure_codec torus_secure_codec.cpp)
target_link_libraries(torus_secure_codec PRIVATE OpenSSL::Crypto)
if(NOT MSVC)
  target_compile_options(torus_secure_codec PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
enable_testing()
add_test(NAME lifting_inverse COMMAND torus_secure_codec selftest)
