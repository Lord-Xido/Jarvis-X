cmake_minimum_required(VERSION 3.12)
project(Moagi3DOperational LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(moagi3d src/moagi3d.cpp)
if(MSVC)
  target_compile_options(moagi3d PRIVATE /W4)
else()
  target_compile_options(moagi3d PRIVATE -Wall -Wextra -Wpedantic)
endif()
find_package(OpenSSL QUIET)
if(OpenSSL_FOUND)
  target_compile_definitions(moagi3d PRIVATE MOAGI_ENABLE_OPENSSL=1)
  target_link_libraries(moagi3d PRIVATE OpenSSL::Crypto)
  message(STATUS "AES-256-GCM seal/unseal enabled via OpenSSL")
else()
  message(STATUS "OpenSSL not available; seal/unseal commands disabled")
endif()
enable_testing()
add_test(NAME engine_selftest COMMAND moagi3d selftest)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli_roundtrip COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:moagi3d>)
  add_test(NAME security_envelope COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_secure.py $<TARGET_FILE:moagi3d>)
endif()
