cmake_minimum_required(VERSION 3.16)
project(jarvis_x_dm3d_typed_ir LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(dm3d_typed_ir dm3d_typed_ir.cpp)
target_include_directories(dm3d_typed_ir PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(dm3d_typed_ir PRIVATE DM3D_IR_INCLUDE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
  target_compile_options(dm3d_typed_ir PRIVATE /O2)
else()
  target_compile_options(dm3d_typed_ir PRIVATE -O3 -Wall -Wextra -Wpedantic)
endif()
