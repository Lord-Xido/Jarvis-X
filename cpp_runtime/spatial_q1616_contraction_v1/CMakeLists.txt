cmake_minimum_required(VERSION 3.16)
project(jarvis_x_spatial_q1616_contraction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(jarvis_x_spatial_q1616 INTERFACE)
target_include_directories(jarvis_x_spatial_q1616 INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(spatial3d_demo src/main.cpp)
target_link_libraries(spatial3d_demo PRIVATE jarvis_x_spatial_q1616)
add_executable(spatial3d_test tests/test_spatial.cpp)
target_link_libraries(spatial3d_test PRIVATE jarvis_x_spatial_q1616)
enable_testing()
add_test(NAME spatial_q1616_test COMMAND spatial3d_test)
