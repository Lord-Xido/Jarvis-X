cmake_minimum_required(VERSION 3.18)
project(JARVISX_V35 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_executable(jarvis_x_v35 jarvis_x_v35_frequency_vm.cpp)
target_compile_options(jarvis_x_v35 PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(jarvis_x_v35 PRIVATE OpenSSL::Crypto)
enable_testing()
add_test(NAME v35_selftest COMMAND jarvis_x_v35 --self-test)

add_test(NAME v35_invalid_negative_frequency COMMAND jarvis_x_v35 --frequency -1)
set_tests_properties(v35_invalid_negative_frequency PROPERTIES WILL_FAIL TRUE)
add_test(NAME v35_invalid_large_frequency COMMAND jarvis_x_v35 --frequency 32)
set_tests_properties(v35_invalid_large_frequency PROPERTIES WILL_FAIL TRUE)
