cmake_minimum_required(VERSION 3.16)
project(jarvisx_perspective3d LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(jarvisx-perspective3d-tests perspective3d_tests.cpp)
if(MSVC)
  target_compile_options(jarvisx-perspective3d-tests PRIVATE /W4 /permissive-)
else()
  target_compile_options(jarvisx-perspective3d-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
enable_testing()
add_test(NAME perspective3d-conformance COMMAND jarvisx-perspective3d-tests)
set_tests_properties(perspective3d-conformance PROPERTIES TIMEOUT 30)
