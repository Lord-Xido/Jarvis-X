cmake_minimum_required(VERSION 3.20)
project(JARVISX_CubePNG_Neural_v6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PNG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(cubepng_neural cubepng_neural.cpp neural_mlp.hpp)
target_link_libraries(cubepng_neural PRIVATE PNG::PNG ZLIB::ZLIB OpenSSL::Crypto)
target_compile_options(cubepng_neural PRIVATE -Wall -Wextra -Wpedantic -Werror)
include(CTest)
if(BUILD_TESTING)
 add_test(NAME predictor_and_residual COMMAND cubepng_neural self-test)
 add_test(NAME lossless_roundtrip_and_corruption COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/roundtrip.sh $<TARGET_FILE:cubepng_neural> ${CMAKE_CURRENT_BINARY_DIR}/fixture.png)
 add_test(NAME untrusted_limits_and_atomic_publish COMMAND bash ${CMAKE_CURRENT_SOURCE_DIR}/tests/security.sh $<TARGET_FILE:cubepng_neural> ${CMAKE_CURRENT_BINARY_DIR}/fixture.png)
endif()
