cmake_minimum_required(VERSION 3.16)
project(jarvisx_spatial_q16 LANGUAGES CXX)
add_executable(spatial_q16 spatial_q16.cpp)
target_compile_features(spatial_q16 PRIVATE cxx_std_20)
set_target_properties(spatial_q16 PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(spatial_q16 PRIVATE /W4 /permissive-)
else()
  target_compile_options(spatial_q16 PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  option(SPATIAL_Q16_SANITIZERS "Enable undefined-behavior sanitization" OFF)
  if(SPATIAL_Q16_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(spatial_q16 PRIVATE -fsanitize=undefined -fno-omit-frame-pointer)
    target_link_options(spatial_q16 PRIVATE -fsanitize=undefined)
  endif()
endif()
enable_testing()
add_test(NAME spatial-q16-selftest COMMAND spatial_q16 selftest)
add_test(NAME spatial-q16-neighborhood-demo COMMAND spatial_q16 demo 30)
set_tests_properties(spatial-q16-selftest PROPERTIES TIMEOUT 120)
