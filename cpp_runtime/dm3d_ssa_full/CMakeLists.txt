cmake_minimum_required(VERSION 3.16)
project(dm3d_full_system VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(dm3d_ssa_codegen src/main.cpp)
target_include_directories(dm3d_ssa_codegen PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(dm3d_ssa_codegen PRIVATE DM3D_SSA_HEADER="${CMAKE_CURRENT_SOURCE_DIR}/include/dm3d_ssa.hpp" DM3D_CXX="${CMAKE_CXX_COMPILER}")
add_executable(dm3d_ssa_tests tests/tests.cpp)
target_include_directories(dm3d_ssa_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(MSVC)
  target_compile_options(dm3d_ssa_codegen PRIVATE /W4 /O2)
  target_compile_options(dm3d_ssa_tests PRIVATE /W4 /O2)
else()
  target_compile_options(dm3d_ssa_codegen PRIVATE -Wall -Wextra -Wpedantic -O2)
  target_compile_options(dm3d_ssa_tests PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
enable_testing()
add_test(NAME dm3d-ssa-unit COMMAND dm3d_ssa_tests)
add_test(NAME dm3d-ssa-e2e COMMAND dm3d_ssa_codegen 4 12 8 "${CMAKE_CURRENT_BINARY_DIR}/e2e")
set_tests_properties(dm3d-ssa-e2e PROPERTIES TIMEOUT 180)
